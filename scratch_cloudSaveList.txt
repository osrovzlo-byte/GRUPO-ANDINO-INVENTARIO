async function cloudSaveList(docName, localList, lastSavedJson, keyFn) {
            if (typeof window.runTransaction !== 'function') {
                return window.setDoc(window.doc(window.db, "lotto_data", docName), { list: JSON.parse(JSON.stringify(localList)) });
            }
            let prev = [];
            try { prev = JSON.parse(lastSavedJson || '[]') || []; } catch (e) { prev = []; }
            const prevMap = new Map();
            prev.forEach(it => { if (it) prevMap.set(keyFn(it), stableStringify(it)); });
            const curMap = new Map();
            localList.forEach(it => { if (it) curMap.set(keyFn(it), it); });
            const upserts = [];
            curMap.forEach((it, k) => { if (prevMap.get(k) !== stableStringify(it)) upserts.push([k, it]); });
            
            const removed = [];
            prevMap.forEach((_, k) => { if (!curMap.has(k)) removed.push(k); });
            if (!upserts.length && !removed.length) {
                console.log(`[CloudSync] No hay cambios reales en ${docName} para subir.`); showToast(`No hay cambios reales detectados para ${docName}`, "info");
                return Promise.resolve(null);
            }

            console.log(`[CloudSync] Subiendo ${docName}: upserts=${upserts.length}, removed=${removed.length}`); showToast(`Subiendo ${docName} a la nube...`, "info");
            
            const ref = window.doc(window.db, "lotto_data", docName);
            try {
                await window.runTransaction(window.db, async (tx) => {
                    const snap = await tx.get(ref);
                    const map = new Map();
                    if (snap.exists()) (snap.data().list || []).forEach(it => { if (it) map.set(keyFn(it), it); });
                    removed.forEach(k => map.delete(k));
                    upserts.forEach(([k, it]) => map.set(k, JSON.parse(JSON.stringify(it))));
                    tx.set(ref, { list: Array.from(map.values()) }); showToast(`Cambios de ${docName} guardados en la nube con Ã©xito!`, "success");
                });
            } catch (err) {
                console.warn(`[Sync] runTransaction failed for ${docName} (${err.message}), falling back to setDoc`);
                return window.setDoc(ref, { list: JSON.parse(JSON.stringify(localList)) });
            }
        }

        function addMovement(movObj) {
            movements.push(movObj);
            if (window.db) {
                window.setDoc(window.doc(window.db, "lotto_movements", movObj.id), movObj).catch(err => {
                    console.error("Error saving movement:", err);
                    handleFirestoreError(err, 'addMovement');
                });
            }
            persistMovementsSoon();
        }

        function saveData() {
            // Depurar eliminados antes de guardar
            if (typeof isEquipmentDeleted === 'function') inventory = inventory.filter(i => !isEquipmentDeleted(i.serial, i.id));
            if (typeof isAgencyDeleted === 'function') agencies = agencies.filter(a => !isAgencyDeleted(a.id, a.code, a.name));

            // Guardado local obligatorio primero
            try { localStorage.setItem('lotto_inventory_v2', JSON.stringify(inventory)); } catch (e) { safeSetLocalStorage('lotto_inventory_v2', JSON.stringify(inventory)); }
            try { localStorage.setItem('lotto_agencies_v2', JSON.stringify(agencies)); } catch (e) { safeSetLocalStorage('lotto_agencies_v2', JSON.stringify(agencies)); }
            persistMovementsSoon();
            try { localStorage.setItem('lotto_maintenance_v2', JSON.stringify(maintenanceLog)); } catch (e) { safeSetLocalStorage('lotto_maintenance_v2', JSON.stringify(maintenanceLog)); }
            try { localStorage.setItem('lotto_categories_v2', JSON.stringify(equipmentCategories)); } catch (e) { safeSetLocalStorage('lotto_categories_v2', JSON.stringify(equipmentCategories)); }
            try { localStorage.setItem('lotto_stock_thresholds_v2', JSON.stringify(stockThresholds)); } catch (e) { safeSetLocalStorage('lotto_stock_thresholds_v2', JSON.stringify(stockThresholds)); }
            try {
                if (typeof deletedEquipmentSerials !== 'undefined') localStorage.setItem('lotto_deleted_serials', JSON.stringify(Array.from(deletedEquipmentSerials)));
                if (typeof deletedEquipmentIds !== 'undefined') localStorage.setItem('lotto_deleted_ids', JSON.stringify(Array.from(deletedEquipmentIds)));
                if (typeof saveDeletedAgenciesTracking === 'function') saveDeletedAgenciesTracking();
            } catch (e) {}

            if (window.db) {
                let currentInv = JSON.stringify(inventory);
                let invChanged = currentInv !== lastSavedInventory;
                console.log('[CloudSync] saveData() — inventario cambió:', invChanged, '| items:', inventory.length);
                if (invChanged) {
                    localStorage.setItem('lotto_pending_inventory_sync', 'true');
                    cloudSaveList('inventory', inventory, lastSavedInventory, invKey)
                        .then(() => { 
                            lastSavedInventory = currentInv; 
                            localStorage.removeItem('lotto_pending_inventory_sync');
                            console.log('[CloudSync] ✅ Inventario subido a la nube exitosamente (' + inventory.length + ' equipos)');
                        })
                        .catch(err => {
                            localStorage.setItem('lotto_pending_inventory_sync', 'true');
                            console.error('[CloudSync] ❌ ERROR subiendo inventario a la nube:', err.code, err.message);
                            handleFirestoreError(err, 'saveInventory');
                        });
                }

                let currentAgencies = JSON.stringify(agencies);
                let agChanged = currentAgencies !== lastSavedAgencies;
                console.log('[CloudSync] saveData() — agencias cambió:', agChanged, '| total:', agencies.length);
                if (agChanged) {
                    localStorage.setItem('lotto_pending_agencies_sync', 'true');
                    cloudSaveList('agencies', agencies, lastSavedAgencies, agKey)
                        .then(() => { 
                            lastSavedAgencies = currentAgencies; 
                            localStorage.removeItem('lotto_pending_agencies_sync');
                            console.log('[CloudSync] ✅ Agencias subidas a la nube exitosamente (' + agencies.length + ' agencias)');
                        })
                        .catch(err => {
                            localStorage.setItem('lotto_pending_agencies_sync', 'true');
                            console.error('[CloudSync] ❌ ERROR subiendo agencias a la nube:', err.code, err.message);
                            handleFirestoreError(err, 'saveAgencies');
                        });
                }

                // (El tracking de agencias eliminadas se sincroniza solo si cambia, en saveDeletedAgenciesTracking)

                let currentMaint = JSON.stringify(maintenanceLog);
                let maintChanged = currentMaint !== lastSavedMaintenance;
                if (maintChanged) {
                    localStorage.setItem('lotto_pending_maintenance_sync', 'true');
                    cloudSaveList('maintenance', maintenanceLog, lastSavedMaintenance, maintKey)
                        .then(() => { 
                            lastSavedMaintenance = currentMaint; 
                            localStorage.removeItem('lotto_pending_maintenance_sync');
                            console.log('[CloudSync] ✅ Mantenimiento subido a la nube exitosamente');
                        })
                        .catch(err => {
                            localStorage.setItem('lotto_pending_maintenance_sync', 'true');
                            console.error('[CloudSync] ❌ ERROR subiendo mantenimiento:', err.code, err.message);
                            handleFirestoreError(err, 'saveMaintenance');
                        });
                }

                let currentSettings = JSON.stringify({ categories: equipmentCategories, thresholds: stockThresholds });
                if (currentSettings !== lastSavedSettings) {
                    window.setDoc(window.doc(window.db, "lotto_data", "settings"), { categories: equipmentCategories, thresholds: stockThresholds })
                        .then(() => { 
                            lastSavedSettings = currentSettings;
                            console.log('[CloudSync] ✅ Configuración subida a la nube exitosamente');
                        })
                        .catch(err => {
                            console.error('[CloudSync] ❌ ERROR subiendo configuración:', err.code, err.message);
                            handleFirestoreError(err, 'saveSettings');
                        });
                }
            }
        }

        // Sincronización manual forzada de agencias con la nube
        function forceSyncAgenciesToCloud() {
            if (!window.db) {
                showToast('Base de datos en la nube no inicializada (Modo Local).', 'warning');
                return;
            }
            if (typeof isAgencyDeleted === 'function') agencies = agencies.filter(a => !isAgencyDeleted(a.id, a.code, a.name));
            showToast('Sincronizando agencias con la nube...', 'info');
            window.setDoc(window.doc(window.db, "lotto_data", "agencies"), { list: agencies })
                .then(() => {
                    lastSavedAgencies = JSON.stringify(agencies);
                    localStorage.removeItem('lotto_pending_agencies_sync');
                    if (typeof saveDeletedAgenciesTracking === 'function') saveDeletedAgenciesTracking();
                    showToast(`¡${agencies.length} agencias sincronizadas exitosamente con la nube!`, 'success');
                })
                .catch(err => {
                    handleFirestoreError(err, 'forceSyncAgencies');
                    if (err && (err.code === 'resource-exhausted' || (err.message && err.message.toLowerCase().includes('quota')))) {
                        showToast('Cuota gratuita diaria de Firebase excedida (429). Puedes usar "Exportar Agencias" para pasarlas en Excel al otro equipo.', 'warning');
                    } else {
                        showToast('Error al sincronizar con la nube: ' + (err.message || 'Verifique conexión'), 'error');
                    }
                });
        }

        /* --- CONTROL Y ALERTAS DE STOCK MÍNIMO --- */
        function checkStockAlerts() {
            const lowStockAlerts = [];
            if (!Array.isArray(inventory)) return lowStockAlerts;
            const usedCategories = new Set(inventory.map(i => (i && i.type) ? String(i.type).trim() : '').filter(Boolean));
            const allCategories = Array.from(new Set([...(equipmentCategories || []), ...usedCategories]));

            allCategories.forEach(cat => {
                if (!cat) return;
                const inWarehouseCount = inventory.filter(i => i && i.type && String(i.type).toLowerCase() === String(cat).toLowerCase() && isWarehouse(i.status)).length;
                const minThreshold = (stockThresholds && stockThresholds[cat] !== undefined) ? stockThresholds[cat] : 3;

                if (inWarehouseCount <= minThreshold) {
                    lowStockAlerts.push({
                        cat,
                        count: inWarehouseCount,
                        min: minThreshold
                    });
                }
            });

            const banner = document.getElementById('stockAlertBanner');
            const textEl = document.getElementById('stockAlertBannerText');
            const badge = document.getElementById('stockAlertBadge');

            if (lowStockAlerts.length > 0) {
                if (banner) banner.classList.remove('hidden');
                if (badge) {
                    badge.innerText = lowStockAlerts.length;
                    badge.classList.remove('hidden');
                }

                const alertMessages = lowStockAlerts.map(a => `Quedan <strong>${a.count} ${a.cat}(s)</strong> en stock (Mín: ${a.min})`);
                if (textEl) {
                    textEl.innerHTML = alertMessages.join(' • ');
                }
            } else {
                if (banner) banner.classList.add('hidden');
                if (badge) badge.classList.add('hidden');
            }

            return lowStockAlerts;
        }

        function triggerStockToastAlerts(affectedTypes = []) {
            affectedTypes.forEach(cat => {
                const inWarehouseCount = inventory.filter(i => i.type.toLowerCase() === cat.toLowerCase() && isWarehouse(i.status)).length;
                const minThreshold = stockThresholds[cat] !== undefined ? stockThresholds[cat] : 3;

                if (inWarehouseCount <= minThreshold) {
                    showToast(`⚠️ ¡Alerta! Quedan ${inWarehouseCount} ${cat}(s) en stock en Depósito (Mínimo: ${minThreshold})`, 'warning');
                }
            });
        }

        function openStockAlertsModal() {
            renderStockThresholdsInputs();
            document.getElementById('stockAlertsModal').classList.remove('hidden');
        }

        function renderStockThresholdsInputs() {
            const container = document.getElementById('stockThresholdsContainer');
            container.innerHTML = '';

            const usedCategories = new Set(inventory.map(i => i.type));
            const allCategories = Array.from(new Set([...equipmentCategories, ...usedCategories]));

            allCategories.forEach(cat => {
                const currentMin = stockThresholds[cat] !== undefined ? stockThresholds[cat] : 3;
                const inWarehouse = inventory.filter(i => i.type.toLowerCase() === cat.toLowerCase() && isWarehouse(i.status)).length;
                const isLow = inWarehouse <= currentMin;

                const row = document.createElement('div');
                row.className = `flex items-center justify-between p-2.5 sm:p-3 rounded-xl border text-xs shadow-sm ${isLow ? 'bg-amber-50/70 border-amber-300' : 'bg-white border-slate-200'}`;

                row.innerHTML = `
                    <div class="flex items-center space-x-2">
                        <div class="p-1.5 ${isLow ? 'bg-amber-100 text-amber-700' : 'bg-slate-100 text-slate-600'} rounded-lg">
                            <i class="${getEquipmentIcon(cat)} text-sm"></i>
                        </div>
                        <div>
                            <span class="font-bold text-slate-800 text-xs sm:text-sm">${cat}</span>
                            <span class="text-[10px] sm:text-[11px] block ${isLow ? 'text-amber-800 font-extrabold' : 'text-slate-500'}">
                                Libre en Depósito: <strong>${inWarehouse}</strong> ${isLow ? '⚠️ (Stock Bajo)' : ''}
                            </span>
                        </div>
                    </div>
                    <div class="flex items-center space-x-1.5">
                        <span class="text-[10px] sm:text-[11px] text-slate-500 font-medium hidden sm:inline">Mínimo:</span>
                        <input type="number" min="0" max="999" data-cat="${cat}" value="${currentMin}" class="w-14 sm:w-16 py-1 px-2 text-center text-xs sm:text-sm font-bold border border-slate-300 rounded-lg focus:ring-2 focus:ring-amber-500 focus:outline-none bg-slate-50">
                    </div>
                `;
                container.appendChild(row);
            });
        }


                function saveStockThresholds(e) {
            e.preventDefault();
            const inputs = document.querySelectorAll('#stockThresholdsContainer input[data-cat]');
            inputs.forEach(input => {
                const cat = input.getAttribute('data-cat');
                const val = parseInt(input.value) || 0;
                stockThresholds[cat] = val;
            });

            addMovement({
                id: newMovId(),
                date: getFormattedNow(),
                type: 'Configuración Stock',
                items: [],
                agency: 'Depósito Central',
                receptorType: 'Configuración',
                receptor: 'Control de Stock',
                owner: 'Grupo Andino',
                user: getActiveUserName(),
                obs: 'Modificación de umbrales mínimos de alerta de stock en inventario.',
                signature: ''
            });

            saveData();
            closeModal('stockAlertsModal');
            renderAll();
            showToast('¡Configuración de alertas de stock guardada exitosamente!');
        }

                function renderQuantitativeReport() {
            const tbody = document.getElementById('quantTableBody');
            tbody.innerHTML = '';

            const categoryStats = [];
            const usedCategories = new Set(inventory.map(i => i.type));
            const allCategories = Array.from(new Set([...equipmentCategories, ...usedCategories]));

            let maxCount = 0;
            let topCategoryName = '--';

            allCategories.forEach(cat => {
                const itemsOfCat = inventory.filter(i => i.type.toLowerCase() === cat.toLowerCase());
                const total = itemsOfCat.length;
                if (total === 0) return;

                const inWarehouse = itemsOfCat.filter(i => isWarehouse(i.status)).length;
                const inAgency = itemsOfCat.filter(i => isAssigned(i.status)).length;
                const inMaintenance = itemsOfCat.filter(i => isMaintenance(i.status)).length;
                const minThreshold = stockThresholds[cat] !== undefined ? stockThresholds[cat] : 3;

                if (total > maxCount) {
                    maxCount = total;
                    topCategoryName = `${cat} (${total})`;
                }

                categoryStats.push({
                    cat,
                    total,
                    inWarehouse,
                    inAgency,
                    inMaintenance,
                    minThreshold,
                    isLow: inWarehouse <= minThreshold
                });
            });

            categoryStats.sort((a, b) => b.total - a.total);

            document.getElementById('quantTotalCount').innerText = inventory.length;
            document.getElementById('quantTypesCount').innerText = categoryStats.length;
            document.getElementById('quantTopCategory').innerText = topCategoryName;

            if (categoryStats.length === 0) {
                tbody.innerHTML = '<tr><td colspan="6" class="p-6 text-center text-slate-400">No hay equipos registrados en el inventario.</td></tr>';
                return;
            }

            categoryStats.forEach(st => {
                const tr = document.createElement('tr');
                tr.className = `hover:bg-slate-100 transition ${st.isLow ? 'bg-amber-50/50' : ''}`;

                const alertBadge = st.isLow ? 
                    `<span class="ml-1.5 px-1.5 py-0.5 text-[9px] sm:text-[10px] font-extrabold bg-amber-100 text-amber-800 border border-amber-300 rounded inline-flex items-center gap-0.5"><i class="fa-solid fa-triangle-exclamation"></i> Mín: ${st.minThreshold}</span>` : '';

                tr.innerHTML = `
                    <td class="p-2.5 sm:p-3 font-semibold text-slate-900 flex items-center space-x-1.5 sm:space-x-2">
                        <div class="p-1 sm:p-1.5 bg-slate-100 rounded-lg text-slate-600 shrink-0">
                            <i class="${getEquipmentIcon(st.cat)} text-xs sm:text-sm"></i>
                        </div>
                        <span class="text-xs sm:text-sm font-bold truncate">${st.cat}</span>
                        ${alertBadge}
                    </td>
                    <td class="p-2.5 sm:p-3 text-center font-bold text-slate-900 text-xs sm:text-sm">
                        <span class="px-2 py-0.5 sm:px-2.5 sm:py-1 bg-slate-100 rounded-full border border-slate-200">${st.total}</span>
                    </td>
                    <td class="p-2.5 sm:p-3 text-center">
                        <span class="px-1.5 py-0.5 rounded font-bold text-xs ${st.isLow ? 'bg-amber-200 text-amber-950 font-black' : st.inWarehouse > 0 ? 'bg-emerald-100 text-emerald-800' : 'text-slate-400'}">${st.inWarehouse}</span>
                    </td>
                    <td class="p-2.5 sm:p-3 text-center">
                        <span class="px-1.5 py-0.5 rounded font-bold text-xs ${st.inAgency > 0 ? 'bg-amber-100 text-amber-800' : 'text-slate-400'}">${st.inAgency}</span>
                    </td>
                    <td class="p-2.5 sm:p-3 text-center">
                        <span class="px-1.5 py-0.5 rounded font-bold text-xs ${st.inMaintenance > 0 ? 'bg-rose-100 text-rose-800' : 'text-slate-400'}">${st.inMaintenance}</span>
                    </td>
                    <td class="p-2.5 sm:p-3 text-right">
                        <button onclick="filterInventoryByType('${st.cat}')" class="px-2 py-1 bg-blue-50 hover:bg-blue-100 text-blue-700 rounded-lg font-semibold text-[10px] sm:text-[11px] border border-blue-200 transition">
                            <i class="fa-solid fa-filter mr-0.5"></i> Filtrar
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

                function renderInventoryTable() {
            const tbody = document.getElementById('inventoryTableBody');
            const emptyMsg = document.getElementById('noInventoryMsg');
            tbody.innerHTML = '';

            const query = document.getElementById('filterSearch').value.toLowerCase();
            const typeFilter = document.getElementById('filterType').value;
            const statusFilter = document.getElementById('filterStatus').value;
            const brandFilter = document.getElementById('filterBrand').value.toLowerCase();

                        const filtered = inventory.filter(item => {
                const matchesQuery = !query || 
                    (item.serial && String(item.serial).toLowerCase().includes(query)) ||
                    (item.type && String(item.type).toLowerCase().includes(query)) ||
                    (item.brand && String(item.brand).toLowerCase().includes(query)) ||
                    (item.model && String(item.model).toLowerCase().includes(query)) ||
                    (item.processor && String(item.processor).toLowerCase().includes(query)) ||
                    (item.agency && String(item.agency).toLowerCase().includes(query)) ||
                    (item.receptor && String(item.receptor).toLowerCase().includes(query)) ||
                    (item.owner && String(item.owner).toLowerCase().includes(query));

                const matchesType = !typeFilter || (item.type && String(item.type).toLowerCase() === typeFilter.toLowerCase());
                const matchesStatus = !statusFilter || 
                    (statusFilter === 'En Depósito' ? isWarehouse(item.status) : 
                     statusFilter === 'Asignado' ? isAssigned(item.status) : 
                     statusFilter === 'En Mantenimiento' ? isMaintenance(item.status) : 
                     item.status === statusFilter);
                const matchesBrand = !brandFilter || (item.brand && String(item.brand).toLowerCase().includes(brandFilter));

                return matchesQuery && matchesType && matchesStatus && matchesBrand;
            });

            if (filtered.length === 0) {
                emptyMsg.classList.remove('hidden');
            } else {
                emptyMsg.classList.add('hidden');
                filtered.forEach(item => {
                    const tr = document.createElement('tr');
                    tr.className = 'hover:bg-slate-50 transition border-b border-slate-100 text-xs sm:text-sm';

        function parseSeriesRange(fromRaw, toRaw) {
            const fromStr = (fromRaw || '').trim().toUpperCase();
            const toStr = (toRaw || '').trim().toUpperCase();
            if (!fromStr || !toStr) return null;

            const matchFrom = fromStr.match(/^(.*?)(\d+)$/);
            const matchTo = toStr.match(/^(.*?)(\d+)$/);

            if (!matchFrom || !matchTo) {
                return { error: 'Los valores inicial y final deben terminar en números correlativos (ej: 6580 al 6590).' };
            }

            const prefixFrom = matchFrom[1];
            const prefixTo = matchTo[1];

            if (prefixFrom !== prefixTo) {
                return { error: 'El prefijo del serial inicial no coincide con el final (ej: BAN-6580 al BAN-6590).' };
            }

            const fromNum = parseInt(matchFrom[2], 10);
            const toNum = parseInt(matchTo[2], 10);

            if (isNaN(fromNum) || isNaN(toNum)) {
                return { error: 'Los números de serie no son válidos.' };
            }

            if (toNum < fromNum) {
                return { error: 'El número final debe ser mayor o igual al número inicial.' };
            }

            const count = (toNum - fromNum) + 1;
            if (count > 500) {
                return { error: `El rango seleccionado (${count} equipos) supera el límite máximo permitido por lote (500 equipos).` };
            }

            const padLen = Math.max(matchFrom[2].length, matchTo[2].length);
            const serials = [];
            for (let i = fromNum; i <= toNum; i++) {
                const numPadded = String(i).padStart(padLen, '0');
                serials.push(`${prefixFrom}${numPadded}`);
            }

            return {
                prefix: prefixFrom,
                fromNum,
                toNum,
                count,
                serials
            };
        }

            const supervisor = document.getElementById('agencySupervisor').value.trim();
            const mapsUrl = document.getElementById('agencyMapsUrl').value.trim();

            if (editingId && editingId !== "undefined") {
                const agIndex = agencies.findIndex(a => {
                    if (!a) return false;
                    return (a.id && a.id === editingId) || (a.code && a.code === editingId) || (a.name && a.name === editingId);
                });
                if (agIndex >= 0) {
                    const targetKey = agKey(agencies[agIndex]);
                    // ELIMINAR CUALQUIER DUPLICADO DE ESTA AGENCIA ANTES DE ACTUALIZAR
                    agencies = agencies.filter((a, idx) => {
                        if (!a) return false;
                        if (agKey(a) !== targetKey) return true;
                        return idx === agIndex;
                    });

                    const dupAgEdit = agencies.find(a => a && a !== agencies[agIndex] && ((a.code || a.id || '').trim().toUpperCase() === code || (a.name || '').trim().toUpperCase() === name.toUpperCase()));
                    if (dupAgEdit) {
                        showToast(`¡Error! Otra agencia ya usa ese código o nombre (${dupAgEdit.name} / ${dupAgEdit.code}).`, 'error');
                        return;
                    }
                    const oldName = agencies[agIndex].name;
                    agencies[agIndex] = {
                        ...agencies[agIndex],
                        id: agencies[agIndex].id || agencies[agIndex].code || code, // ensure ID is preserved or created
                        code,
                        name,
                        zone,
                        address,
                        owner,
                        phone,
                        supervisor,
                        mapsUrl
                    };

                    if (oldName !== name) {
                        inventory.forEach(item => {
                            if (item.agency === oldName) {
                                item.agency = name;
                            }
                        });
                    }
                    if (typeof movements !== 'undefined') {
                        let movChanged = false;
                        movements.forEach(mov => {
                            if (mov.agency === oldName) {
                                mov.agency = name;
                                movChanged = true;
                            
                                    if (window.db) {
                                        window.setDoc(window.doc(window.db, "lotto_movements", mov.id), mov).catch(e => console.error(e));
                                    }
                                }
                        });
                        if (movChanged && typeof persistMovementsSoon === 'function') {
                            persistMovementsSoon();
                        }
                    }

                    if (typeof unmarkAgencyAsDeleted === 'function') {
                        unmarkAgencyAsDeleted(agencies[agIndex].id, code, name);
                    }

                    addMovement({
                        id: newMovId(),
                        date: getFormattedNow(),
                        type: 'Edición Agencia',
                        items: [],
                        agency: name,
                        receptorType: 'Catálogo Agencias',
                        receptor: owner || supervisor || '-',
                        owner: owner || '-',
                        user: getActiveUserName(),
                        obs: `Actualización de datos de agencia: ${name} (Cód: ${code}). Supervisor: ${supervisor || '-'}. Tel: ${phone || '-'}`,
                        signature: ''
                    });

                    showToast('¡Información de la agencia actualizada con éxito!', 'success');
                }
            } else {
                const dupAg = agencies.find(a => a && ((a.code || a.id || '').trim().toUpperCase() === code || (a.name || '').trim().toUpperCase() === name.toUpperCase()));
                if (dupAg) {
                    showToast(`¡Error! Ya existe una agencia con ese código o nombre (${dupAg.name} / ${dupAg.code}).`, 'error');
                    return;
                }
                const newAgency = {
                    id: code,
                    code,
                    name,
                    zone,
                    address,
                    owner,
                    phone,
                    supervisor,
                    mapsUrl
                };

                if (typeof unmarkAgencyAsDeleted === 'function') {

                function deleteAgency(id) {
            if (!checkPermission(['admin'])) return;
            const targetKey = String(id || '').trim();
            const ag = agencies.find(a => {
                if (!a) return false;
                if (a.id && targetKey && String(a.id).trim() === targetKey) return true;
                if (a.code && targetKey && String(a.code).trim() === targetKey) return true;
                if (a.name && targetKey && String(a.name).trim().toLowerCase() === targetKey.toLowerCase()) return true;
                return false;
            });
            if (!ag) {
                showToast('No se encontró la agencia a eliminar.', 'warning');
                return;
            }

            showConfirm(
                '¿Eliminar Agencia?',
                `¿Desea eliminar la agencia "${ag.name}"? Los equipos mantendrán su asignación.`,
                () => {
                    const targetId = ag.id;
                    const targetCode = ag.code;
                    const targetName = ag.name;

                    // 1. Marcar como eliminada de forma persistente
                    if (typeof markAgencyAsDeleted === 'function') {
                        markAgencyAsDeleted(targetId, targetCode, targetName);
                    }

                    // 2. Registrar en auditoría de movimientos
                    addMovement({
                        id: newMovId(),
                        date: getFormattedNow(),
                        type: 'Eliminación Agencia',
                        items: [],
                        agency: targetName,
                        receptorType: 'Catálogo Agencias',
                        receptor: ag.owner || '-',
                        owner: ag.owner || '-',
                        user: getActiveUserName(),
                        obs: `Eliminación de agencia del catálogo: ${targetName} (Cód: ${targetCode || targetId})`,
                        signature: ''
                    });

                    // 3. Remover de la lista en memoria
                    agencies = agencies.filter(a => {
                        if (!a) return false;
                        if (typeof isAgencyDeleted === 'function' && isAgencyDeleted(a.id, a.code, a.name)) return false;
                        
                        // Exact match by ID is safest
                        if (targetId && a.id && String(a.id).trim() === String(targetId).trim()) return false;
                        
                        // Fallbacks if ID was missing
                        if (!targetId && targetCode && a.code && String(a.code).trim() === String(targetCode).trim()) return false;
                        if (!targetId && !targetCode && targetName && a.name && String(a.name).trim().toLowerCase() === String(targetName).trim().toLowerCase()) return false;
                        
                        return true;
                    });

                    saveData();
                    populateAgencyDropdowns();
                    renderAgenciesTable();
                    updateDashboardStats();
                    showToast('Agencia eliminada del catálogo permanentemente.', 'success');
                }
            );
        }

        function processDispatch(e) {
            if (e && e.preventDefault) e.preventDefault();
            if (!checkPermission(['admin', 'tech_head', 'office_exit'])) return;

            const equipId = document.getElementById('dispatchEquipmentId').value;
            const item = inventory.find(i => String(i.id) === String(equipId) || String(i.serial) === String(equipId));
            if (!item) {
                showToast('Error: No se encontró el equipo seleccionado.', 'error');
                return;
            }

            if (!isWarehouse(item.status)) {
                showToast('El equipo seleccionado ya no está disponible en Depósito Central.', 'warning');
                closeModal('dispatchModal');
                return;
            }

            const agency = document.getElementById('dispatchAgencySelect').value;
            if (!agency) {
                showToast('Por favor seleccione una agencia de destino.', 'warning');
                document.getElementById('dispatchAgencySelect').focus();
                return;
            }

            const targetType = document.getElementById('dispatchTargetType').value;
            let owner = document.getElementById('dispatchOwnerName').value.trim();
            if (!owner) {
                const ag = agencies.find(a => a.name === agency);
                owner = ag ? (ag.owner || 'Titular de Agencia') : 'Titular de Agencia';
            }

            let receptor = targetType === 'Dueño Directo' ? '-' : document.getElementById('dispatchReceptorName').value.trim();
            if (targetType !== 'Dueño Directo' && !receptor) {
                receptor = owner;
            }

            const obs = document.getElementById('dispatchObs').value.trim();

            if (isCanvasBlank(canvas)) {
                showToast('Por favor, solicite la firma del receptor antes de continuar.', 'warning');
                return;
            }

            try {
                const signatureData = canvas.toDataURL('image/jpeg', 0.5);
                const now = getFormattedNow();
                const dispatchedType = item.type;

                item.status = 'Asignado';
                item.agency = agency;
                item.owner = owner;
                item.receptor = receptor;
                item.dateOut = now;

                const movId = newMovId();
                const movObj = {
                    id: movId,
                    date: now,
                    type: 'Salida / Asignación',
                    items: [{ type: item.type, brandModel: `${item.brand} ${item.model}`, serial: item.serial, color: item.color, ram: item.ram || '', disk: item.disk || '', processor: item.processor || '' }],
                    agency,
                    receptorType: targetType,
                    receptor,
                    owner,
                    user: getActiveUserName(),
                    obs,
                    signature: signatureData
                };

                addMovement(movObj);
                saveData();
                closeModal('dispatchModal');
                renderAll();

                showToast('¡Despacho completado con éxito!');
                try {
                    triggerStockToastAlerts([dispatchedType]);
                } catch (errAlert) {
                    console.warn('Alerta de stock:', errAlert);
                }
                viewReceipt(movId);
            } catch (err) {
                console.error('Error al procesar despacho:', err);
                showToast('Error al procesar despacho: ' + err.message, 'error');
            }
        }

        function handleExcelFileSelect(e) {
            const file = e.target.files[0];
            if (!file) return;

            document.getElementById('excelFileName').innerText = file.name;
            showToast("Procesando archivo de Excel...", "info");

            const reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    const data = new Uint8Array(evt.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheetName = workbook.SheetNames[0];
                    const worksheet = workbook.Sheets[firstSheetName];
                    const rawRows = XLSX.utils.sheet_to_json(worksheet, { defval: '' });

                    if (rawRows.length === 0) {
                        showToast('El archivo de Excel parece estar vacío.', 'warning');
                        return;
                    }

                    parsedExcelRows = [];
                    const detectedAgencies = new Set();
                    const now = getFormattedNow();

                    rawRows.forEach((r, idx) => {
                        const getVal = (keys) => {
                            for (let k of keys) {
                                for (let rk in r) {
                                    if (rk.toLowerCase().trim() === k.toLowerCase()) return String(r[rk]).trim();
                                }
                            }
                            return '';
                        };

                        const serial = getVal(['serial', 's/n', 'numero de serial', 'número de serie', 'sn', 'código', 'codigo']).toUpperCase();
                        if (!serial) return;

                        // REGLA ESTRICTA DE CATEGORÍAS: Si no existe en el catálogo, colocar estrictamente en 'Otro'
                        const rawType = getVal(['tipo', 'categoria', 'categoría', 'equipo', 'clase']);
                        const cleanRawType = (rawType || '').trim();
                        const matchedCat = equipmentCategories.find(c => c.toLowerCase().trim() === cleanRawType.toLowerCase());
                        
                        let type = 'Otro';
                        let wasFallback = false;
                        if (matchedCat) {
                            type = matchedCat;
                        } else {
                            type = 'Otro';
                            if (cleanRawType && cleanRawType.toLowerCase() !== 'otro') {
                                wasFallback = true;
                            }
                        }

                        const brand = getVal(['marca', 'brand']) || 'Genérico';
                        const model = getVal(['modelo', 'model']) || 'Estándar';
                        const processor = getVal(['procesador', 'cpu', 'processor']) || '';
                        const ram = getVal(['ram', 'memoria', 'memoriaram', 'memoria ram']) || '';
                        const disk = getVal(['discoduro', 'disco duro', 'disco', 'almacenamiento', 'ssd', 'hdd']) || '';
                        const color = getVal(['color']) || 'No especificado';
                        const agency = getVal(['agencia', 'ubicacion', 'ubicación', 'punto de venta', 'agencia asignada']);
                        const receptor = getVal(['receptor', 'subreceptor', 'responsable']) || '-';
                        const owner = getVal(['dueño', 'dueno', 'titular', 'propiedad']) || '-';
                        const statusRaw = getVal(['estado', 'status']);
                        
                        let dateIn = getVal(['fechaentrada', 'fecha entrada', 'fechadeingreso', 'fecha ingreso']) || now;
                        let dateOut = getVal(['fechasalida', 'fecha salida', 'fechadedespacho', 'fecha despacho']) || '-';
                        const obs = getVal(['observaciones', 'observacion', 'notas', 'nota', 'detalles']) || '';

                        const hasAgency = agency && agency !== '-' && agency.toLowerCase() !== 'ninguna' && agency.toLowerCase() !== 'depósito' && agency.toLowerCase() !== 'deposito';
                        
                        let finalStatus = 'En Depósito';
                        if (statusRaw && (statusRaw.toLowerCase().includes('manten') || statusRaw.toLowerCase().includes('taller'))) {
                            finalStatus = 'En Mantenimiento';
                        } else if (hasAgency || (statusRaw && statusRaw.toLowerCase().includes('asig'))) {
                            finalStatus = 'Asignado';
                        }

                        if (hasAgency && dateOut === '-') {
                            dateOut = now;
                        }

                        if (hasAgency && !agencies.some(a => a.name.toLowerCase().trim() === agency.toLowerCase().trim())) {
                            detectedAgencies.add(agency);
                        }

                        parsedExcelRows.push({
                            id: Date.now() + idx + Math.random().toString().slice(2, 6),
                            type,
                            rawType: cleanRawType,
                            wasFallback,
                            brand,
                            model,
                            processor,
                            ram,
                            disk,
                            color,
                            serial,
                            agency: hasAgency ? agency : '-',
                            receptor: hasAgency ? receptor : '-',
                            owner: hasAgency ? owner : '-',
                            status: finalStatus,
                            dateIn,
                            dateOut,
                            obs
                        });
                    });

                    if (parsedExcelRows.length === 0) {
                        showToast('No se pudieron extraer filas con número de Serial válido.', 'warning');
                        return;
                    }

                    const previewSummary = document.getElementById('excelPreviewSummary');
                    const agenciesSummary = document.getElementById('excelAgenciesSummary');
                    const tbody = document.getElementById('excelPreviewTbody');

                    const fallbackCount = parsedExcelRows.filter(r => r.wasFallback).length;
                    previewSummary.innerHTML = `${parsedExcelRows.length} equipos listos ${fallbackCount > 0 ? '<span class="text-amber-600 font-normal">(' + fallbackCount + ' asignados a "Otro")</span>' : ''}`;
                    agenciesSummary.innerText = `${detectedAgencies.size} agencia(s) nueva(s) a incluir`;

                    tbody.innerHTML = '';
                    parsedExcelRows.slice(0, 15).forEach((row, i) => {
                        const tr = document.createElement('tr');
                        const specInfo = [row.processor, row.ram, row.disk].filter(Boolean).join(' / ');
                        tr.innerHTML = `
                            <td class="p-2 font-mono text-slate-400">${i + 1}</td>
                            <td class="p-2">
                                <span class="font-semibold text-slate-800">${row.type}</span>
                                ${row.wasFallback ? `<span class="block text-[9px] text-amber-700 bg-amber-50 rounded px-1 w-fit mt-0.5" title="Categoría original: ${row.rawType}">Reasignado a Otro</span>` : ''}
                            </td>
                            <td class="p-2 text-slate-700">
                                <div class="font-medium">${row.brand} ${row.model}</div>
                                ${specInfo ? `<div class="text-[10px] text-indigo-600 font-mono">${specInfo}</div>` : ''}
                            </td>
                            <td class="p-2 font-mono font-bold text-emerald-700">${row.serial}</td>
                            <td class="p-2 text-slate-800">${row.agency}</td>
                        `;
                        tbody.appendChild(tr);
                    });

                    document.getElementById('excelPreviewContainer').classList.remove('hidden');

                    const btn = document.getElementById('btnConfirmExcelImport');
                    btn.disabled = false;
                    btn.className = "px-5 py-2 text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-700 cursor-pointer rounded-lg shadow transition flex items-center space-x-1.5";

                    showToast(`Excel procesado: ${parsedExcelRows.length} equipos analizados.`, "success");
                } catch (err) {
                    console.error("Excel parse error:", err);
                    showToast('Error al leer el archivo Excel. Verifique que no esté dañado.', 'error');
                }
            };
            reader.readAsArrayBuffer(file);
        }

        function executeExcelImport() {
            if (parsedExcelRows.length === 0) return;

            let addedCount = 0;
            let updatedCount = 0;
            let newAgenciesCreated = 0;

            const now = getFormattedNow();

            parsedExcelRows.forEach(row => {
                // Registrar agencias que no existan
                if (row.agency !== '-' && !agencies.some(a => a.name.toLowerCase().trim() === row.agency.toLowerCase().trim())) {
                    const newCode = nextFreeAgencyCode();
                    agencies.push({
                        id: newCode,
                        code: newCode,
                        name: row.agency,
                        zone: 'Importada por Excel',
                        address: 'Registrada automáticamente vía Excel',
                        owner: row.owner !== '-' ? row.owner : 'Por Especificar',
                        phone: 'Sin teléfono',
                        supervisor: 'Por Asignar'
                    });
                    newAgenciesCreated++;
                }

                // REGLA ESTRICTA: NO crear nuevas categorías. Las no encontradas ya vienen como 'Otro'.
                // NO se ejecuta equipmentCategories.push(...)

                const existingIdx = inventory.findIndex(i => normSerialKey(i.serial) === normSerialKey(row.serial));
                if (existingIdx >= 0) {
                    const prev = inventory[existingIdx];
                    inventory[existingIdx] = {
                        ...prev,
                        type: row.type,
                        brand: row.brand || prev.brand,
                        model: row.model || prev.model,
                        processor: row.processor || prev.processor || '',
                        ram: row.ram || prev.ram || '',
                        disk: row.disk || prev.disk || '',
                        color: row.color || prev.color || '',
                        status: row.status || prev.status,
                        agency: row.agency !== '-' ? row.agency : prev.agency,
                        receptor: row.receptor !== '-' ? row.receptor : prev.receptor,
                        owner: row.owner !== '-' ? row.owner : prev.owner,
                        dateIn: row.dateIn || prev.dateIn,
                        dateOut: row.dateOut !== '-' ? row.dateOut : prev.dateOut,
                        obs: row.obs || prev.obs || ''
                    };
                    updatedCount++;
                } else {
                    inventory.push({
                        id: row.id,
                        type: row.type,
                        brand: row.brand,
                        model: row.model,
                        processor: row.processor || '',
                        ram: row.ram || '',
                        disk: row.disk || '',
                        color: row.color || '',
                        serial: row.serial,
                        status: row.status,
                        agency: row.agency,
                        receptor: row.receptor,
                        owner: row.owner,
                        dateIn: row.dateIn,
                        dateOut: row.dateOut,
                        obs: row.obs || ''
                    });
                    addedCount++;
                }
                if (typeof unmarkEquipmentAsDeleted === 'function') {
                    unmarkEquipmentAsDeleted(row.serial, row.id || (existingIdx >= 0 ? inventory[existingIdx].id : ''));
                }
            });

            addMovement({
                id: newMovId(),
                date: now,
                type: 'Importación Masiva Excel',
                items: [{
                    type: 'Carga Masiva Excel',
                    brandModel: `Alimentación Inventario (${parsedExcelRows.length} equipos)`,
                    serial: `EXCEL-${Date.now().toString().slice(-6)}`,
                    color: 'N/A'
                }],
                agency: 'Depósito / Agencias Importadas',
                receptorType: 'Carga Masiva',
                receptor: 'Administrador',
                owner: 'Grupo Andino',
                user: getActiveUserName(),
                obs: `Carga masiva Excel realizada por ${getActiveUserName()}: ${addedCount} nuevos, ${updatedCount} actualizados, ${newAgenciesCreated} agencias creadas. Categorías no reconocidas asignadas a 'Otro'.`,
                signature: ''
            });

            saveData();
            populateTypeDropdowns();
            populateAgencyDropdowns();
            renderAll();
            closeModal('excelImportModal');

            showToast(`¡Inventario alimentado con éxito! ${addedCount} nuevos equipos, ${updatedCount} actualizados.`, 'success');
        }

        function escapeTagHtml(str) {
            if (!str) return '';
            return String(str)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#39;');
        }

        function updatePrintPageStyle(preset) {
            let styleEl = document.getElementById('dynamicTagPrintStyle');
            if (!styleEl) {
                styleEl = document.createElement('style');
                styleEl.id = 'dynamicTagPrintStyle';
                document.head.appendChild(styleEl);

            let w = '36mm';

        function renderTagCodes(item, preset) {
            if (!item || !item.serial) return;
            const serial = String(item.serial).trim();

            // Renderizado Barcode 1D (Code128)
            try {
                let bWidth = 1.0;
                let bHeight = 24;
                if (preset === '36x27') {
                    bWidth = 1.0;
                    bHeight = 24;
                } else if (preset === '57x19') {
                    bWidth = 1.1;
                    bHeight = 25;
                } else if (preset === '50x25') {
                    bWidth = 1.1;
                    bHeight = 30;
                } else if (preset === 'full') {
                    bWidth = 1.8;
                    bHeight = 50;
                }

                JsBarcode("#tagBarcode", serial, {
                    format: "CODE128",
                    width: bWidth,
                    height: bHeight,
                    displayValue: false,
                    margin: 0
                });
            } catch (e) {
                console.error("Barcode gen error", e);
            }

            // Renderizado QR Code 2D
            const qrContainer = document.getElementById('tagQRCode');
            if (qrContainer) {
                qrContainer.innerHTML = '';
                try {
                    let qrPx = 42;
                    if (preset === '36x27') {
                        qrPx = 42;
                    } else if (preset === '57x19') {
                        qrPx = 52;
                    } else if (preset === '50x25') {
                        qrPx = 52;
                    } else if (preset === 'full') {
                        qrPx = 100;
                    }

                    new QRCode(qrContainer, {
                        text: serial,
                        width: qrPx,
                        height: qrPx,
                        correctLevel: QRCode.CorrectLevel.M
                    });
                } catch (e) {
                    console.error("QR code gen error", e);
                }
            }
        }

        function initSignatureCanvases() {
            canvas = document.getElementById('sigCanvas');
            batchCanvas = document.getElementById('batchSigCanvas');
            returnCanvas = document.getElementById('returnSigCanvas');

            [canvas, batchCanvas, returnCanvas].forEach(cnv => {
                if (!cnv) return;

                cnv.addEventListener('mousedown', startDrawing);
                cnv.addEventListener('mousemove', draw);
                cnv.addEventListener('mouseup', stopDrawing);
                cnv.addEventListener('mouseleave', stopDrawing);

                cnv.addEventListener('touchstart', (e) => {
                    e.preventDefault();
                    startDrawing(e);
                }, { passive: false });

                cnv.addEventListener('touchmove', (e) => {
                    e.preventDefault();
                    draw(e);
                }, { passive: false });

                cnv.addEventListener('touchend', (e) => {
                    e.preventDefault();
                    stopDrawing();
                }, { passive: false });
            });
        }

        function resizeCanvas(cnv) {
            if (!cnv) return;
            const rect = cnv.getBoundingClientRect();
            if (rect.width === 0 || rect.height === 0) return;

            const dpr = window.devicePixelRatio || 1;
            cnv.width = rect.width * dpr;
            cnv.height = rect.height * dpr;
            
            const context = cnv.getContext('2d');
            context.scale(dpr, dpr);
            context.lineWidth = 2.5;
            context.lineCap = 'round';
            context.lineJoin = 'round';
            context.strokeStyle = currentSigColor;
        }

        function setCanvasColor(color) {
            currentSigColor = color;
            if (ctx) ctx.strokeStyle = currentSigColor;
        }

        function getCanvasPos(cnv, e) {
            const rect = cnv.getBoundingClientRect();
            let clientX = e.clientX;
            let clientY = e.clientY;

            if (e.touches && e.touches.length > 0) {
                clientX = e.touches[0].clientX;
                clientY = e.touches[0].clientY;
            }

            return {
                x: clientX - rect.left,
                y: clientY - rect.top
            };
        }

        function startDrawing(e) {
            isDrawing = true;
            const cnv = e.target;
            const context = cnv.getContext('2d');
            const pos = getCanvasPos(cnv, e);
            context.beginPath();
            context.moveTo(pos.x, pos.y);
        }

        function draw(e) {
            if (!isDrawing) return;
            const cnv = e.target;
            const context = cnv.getContext('2d');
            const pos = getCanvasPos(cnv, e);
            context.lineTo(pos.x, pos.y);
            context.stroke();

            if (cnv.id === 'returnSigCanvas') {
                checkReturnValidation();
            }
        }

        function stopDrawing() {
            isDrawing = false;
        }

        function clearCanvasFull(cnv) {
            if (!cnv) return;
            const context = cnv.getContext('2d');
            context.save();
            context.setTransform(1, 0, 0, 1, 0, 0);
            context.clearRect(0, 0, cnv.width, cnv.height);
            context.restore();
        }

        function clearSignature() {
            if (canvas) clearCanvasFull(canvas);
        }

        function clearBatchSignature() {
            if (batchCanvas) clearCanvasFull(batchCanvas);
        }

        function clearReturnSignature() {
            if (returnCanvas) {
                clearCanvasFull(returnCanvas);
                checkReturnValidation();
            }
        }

        function isCanvasBlank(cnv) {
            if (!cnv) return true;
            const context = cnv.getContext('2d');
            const pixelBuffer = new Uint32Array(
                context.getImageData(0, 0, cnv.width, cnv.height).data.buffer
            );
            return !pixelBuffer.some(color => color !== 0);
        }

        // Tabla estándar CRC-32 para empaquetado ZIP sin dependencias
        const crc32Table = new Uint32Array(256);
        for (let i = 0; i < 256; i++) {
            let c = i;
            for (let k = 0; k < 8; k++) {
                c = ((c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1));
            }
            crc32Table[i] = c >>> 0;
        }

        function calculateCrc32(buf) {
            let crc = 0xFFFFFFFF;
            for (let i = 0; i < buf.length; i++) {
                crc = (crc >>> 8) ^ crc32Table[(crc ^ buf[i]) & 0xFF];
            }
            return (crc ^ 0xFFFFFFFF) >>> 0;
        }

        class SimpleZipBrowser {
            constructor() {
                this.files = [];
            }

            addFile(name, data) {
                let uint8;
                if (typeof data === 'string') {
                    uint8 = new TextEncoder().encode(data);
                } else if (data instanceof Uint8Array) {
                    uint8 = data;
                } else if (data instanceof ArrayBuffer) {
                    uint8 = new Uint8Array(data);
                } else {
                    uint8 = new Uint8Array(data);
                }
                this.files.push({ name, data: uint8, crc: calculateCrc32(uint8) });
            }

            generate() {
                const localParts = [];
                const centralParts = [];
                let offset = 0;
                const textEnc = new TextEncoder();

                for (const file of this.files) {
                    const nameBytes = textEnc.encode(file.name);
                    const dataLen = file.data.length;

                    // Encabezado local de archivo (30 bytes + nombre)
                    const lh = new Uint8Array(30 + nameBytes.length);
                    const dvLh = new DataView(lh.buffer);
                    dvLh.setUint32(0, 0x04034b50, true); // Signatura PK\x03\x04
                    dvLh.setUint16(4, 20, true);         // Versión necesaria (2.0)
                    dvLh.setUint16(6, 0x0800, true);     // Flag UTF-8
                    dvLh.setUint16(8, 0, true);          // Método 0 (Sin compresión / Stored)
                    dvLh.setUint16(10, 0, true);
                    dvLh.setUint16(12, 0, true);
                    dvLh.setUint32(14, file.crc, true);
                    dvLh.setUint32(18, dataLen, true);
                    dvLh.setUint32(22, dataLen, true);
                    dvLh.setUint16(26, nameBytes.length, true);
                    dvLh.setUint16(28, 0, true);
                    lh.set(nameBytes, 30);

                    // Encabezado del directorio central (46 bytes + nombre)
                    const ch = new Uint8Array(46 + nameBytes.length);
                    const dvCh = new DataView(ch.buffer);
                    dvCh.setUint32(0, 0x02014b50, true); // Signatura PK\x01\x02
                    dvCh.setUint16(4, 20, true);         // Versión hecha por
                    dvCh.setUint16(6, 20, true);         // Versión requerida
                    dvCh.setUint16(8, 0, true);          // Flag
                    dvCh.setUint16(8, 0x0800, true);     // Flag UTF-8
                    dvCh.setUint16(10, 0, true);         // Método 0 (Stored)
                    dvCh.setUint16(12, 0, true);         // Mod time
                    dvCh.setUint16(14, 0, true);         // Mod date
                    dvCh.setUint32(16, file.crc, true);  // CRC-32
                    dvCh.setUint32(20, dataLen, true);   // Tamaño comprimido
                    dvCh.setUint32(24, dataLen, true);   // Tamaño descomprimido
                    dvCh.setUint16(28, nameBytes.length, true); // Longitud nombre archivo
                    dvCh.setUint16(30, 0, true);         // Longitud campo extra
                    dvCh.setUint16(32, 0, true);         // Longitud comentario
                    dvCh.setUint16(34, 0, true);         // Número de disco
                    dvCh.setUint16(36, 0, true);         // Atributos internos
                    dvCh.setUint32(38, 0, true);         // Atributos externos
                    dvCh.setUint32(42, offset, true);    // Offset relativo del encabezado local
                    ch.set(nameBytes, 46);

                    localParts.push(lh, file.data);
                    centralParts.push(ch);
                    offset += lh.length + file.data.length;
                }

                const cdOffset = offset;
                let cdSize = 0;
                for (const ch of centralParts) cdSize += ch.length;

                // Registro fin de directorio central (22 bytes)
                const eocd = new Uint8Array(22);
                const dvEocd = new DataView(eocd.buffer);
                dvEocd.setUint32(0, 0x06054b50, true); // Signatura PK\x05\x06
                dvEocd.setUint16(4, 0, true);
                dvEocd.setUint16(6, 0, true);
                dvEocd.setUint16(8, this.files.length, true);
                dvEocd.setUint16(10, this.files.length, true);
                dvEocd.setUint32(12, cdSize, true);
                dvEocd.setUint32(16, cdOffset, true);
                dvEocd.setUint16(20, 0, true);

                const totalSize = offset + cdSize + 22;
                const result = new Uint8Array(totalSize);
                let cur = 0;
                for (const p of [...localParts, ...centralParts, eocd]) {
                    result.set(p, cur);
                    cur += p.length;
                }
                return result;
            }
        }

        function base64ToUint8Array(b64) {
            const raw = atob(b64.replace(/^data:image\/\w+;base64,/, ''));
            const u8 = new Uint8Array(raw.length);
            for (let i = 0; i < raw.length; i++) {
                u8[i] = raw.charCodeAt(i);
            }
            return u8;
        }

        function escapeWordXml(str) {
            if (!str) return '';
            return String(str)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&apos;');
        }

                    const ram = it.ram || (invItem ? invItem.ram : '');
                    const disk = it.disk || (invItem ? invItem.disk : '');
                    const specs = [ram ? `RAM: ${ram}` : '', disk ? `Disco: ${disk}` : ''].filter(Boolean).join(', ');
                    return `• ` + it.type + ` ` + (it.brandModel || '') + (specs ? ` [${specs}]` : '') + ` (S/N: ` + it.serial + `)`;
                }).join('\n');

        function initCameraWithPermissions(selectedDeviceId = null) {
            const statusBox = document.getElementById('scannerStatus');
            const selectorContainer = document.getElementById('cameraSelectorContainer');

            try {
                html5QrCodeScanner = new Html5Qrcode("qr-reader", {
                    verbose: false,
                    experimentalFeatures: {
                        useBarCodeDetectorIfSupported: true
                    }
                });
            } catch(e) {
                console.error("Html5Qrcode init error", e);
            }

            const config = { 
                fps: 20, 
                qrbox: (viewfinderWidth, viewfinderHeight) => {
                    const w = Math.floor(viewfinderWidth * 0.85);
                    const h = Math.floor(viewfinderHeight * 0.65);
                    return { width: Math.max(w, 240), height: Math.max(h, 160) };
                },
                aspectRatio: 1.333333
            };

            Html5Qrcode.getCameras().then(devices => {
                if (devices && devices.length > 0) {
                    availableCamerasList = devices;

                    if (selectorContainer) {
                        if (devices.length > 1) {
                            let selectHtml = `<div class="flex items-center space-x-1.5 mb-2"><i class="fa-solid fa-camera-rotate text-emerald-600 text-xs"></i><select id="cameraSelect" onchange="switchCamera(this.value)" class="w-full py-1.5 px-2.5 text-xs border border-slate-300 rounded-lg bg-slate-50 font-semibold text-slate-700 focus:ring-2 focus:ring-emerald-500 focus:outline-none shadow-sm">`;
                            devices.forEach((dev, idx) => {
                                const l = (dev.label || '').toLowerCase();
                                const isBack = l.includes('back') || l.includes('trasera') || l.includes('environment') || l.includes('rear') || idx === devices.length - 1;
                                const isSelected = selectedDeviceId ? (selectedDeviceId === dev.id) : isBack;
                                selectHtml += `<option value="${dev.id}" ${isSelected ? 'selected' : ''}>Cámara ${idx + 1}: ${dev.label || (isBack ? 'Trasera Principal' : 'Frontal')}</option>`;
                            });
                            selectHtml += `</select></div>`;
                            selectorContainer.innerHTML = selectHtml;
                            selectorContainer.classList.remove('hidden');
                        } else {
                            selectorContainer.classList.add('hidden');
                        }
                    }

                    let targetCameraId = selectedDeviceId;
                    if (!targetCameraId) {
                        const backCam = devices.find(d => {
                            const l = (d.label || '').toLowerCase();
                            return l.includes('back') || l.includes('trasera') || l.includes('environment') || l.includes('rear');
                        });
                        targetCameraId = backCam ? backCam.id : devices[devices.length - 1].id;
                    }

                    html5QrCodeScanner.start(
                        targetCameraId,
                        config,
                        onScanSuccess,
                        onScanFailure
                    ).then(() => {
                        statusBox.className = "p-2.5 bg-emerald-50 text-emerald-800 rounded-lg text-xs font-semibold text-center border border-emerald-200";
                        statusBox.innerHTML = `<i class="fa-solid fa-camera text-emerald-600 mr-1"></i> Cámara Activa. Apunte al código de barras...`;
                    }).catch(err => {
                        console.warn("Start camera ID failed, trying facingMode environment fallback", err);
                        startCameraFallback(config);
                    });
                } else {
                    startCameraFallback(config);
                }
            }).catch(err => {
                console.warn("getCameras error or permission pending:", err);
                startCameraFallback(config);
            });
        }

        function requestCameraPermissionsAndStart() {
            stopScanner();
            const statusBox = document.getElementById('scannerStatus');
            statusBox.className = "p-2.5 bg-slate-50 text-slate-700 rounded-lg text-xs font-semibold text-center border border-slate-200";
            statusBox.innerHTML = `<i class="fa-solid fa-circle-notch fa-spin text-emerald-600 mr-1.5"></i> Solicitando permiso... Presiona 'Permitir' en tu pantalla.`;

            if (navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
                navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" } })
                    .then(stream => {
                        stream.getTracks().forEach(track => track.stop());
                        initCameraWithPermissions();
                    })
                    .catch(err => {
                        console.warn("GetUserMedia permission error:", err);
                        showToast("Acceso a la cámara denegado. Verifique permisos en su navegador.", "error");
                        statusBox.innerHTML = `Permiso denegado por el navegador. Permita la cámara en los ajustes del sitio o seleccione una foto.`;
                    });
            } else {
                initCameraWithPermissions();
            }
        }

        function switchCamera(deviceId) {
            stopScanner();
            initCameraWithPermissions(deviceId);
        }

        function onScanSuccess(decodedText, decodedResult) {
            playScanSound();
            const scannedCode = decodedText.trim().toUpperCase();

            if (html5QrCodeScanner) {
                html5QrCodeScanner.stop().then(() => {
                    html5QrCodeScanner.clear();
                    html5QrCodeScanner = null;
                }).catch(err => console.error(err));
            }

            closeModal('qrScannerModal');

            if (activeScanTarget === 'addSerial') {
                const input = document.getElementById('addSerial');
                if (input) {
                    input.value = scannedCode;
                    input.focus();
                }
            } else if (activeScanTarget === 'batchEquipSearchInput') {
                const input = document.getElementById('batchEquipSearchInput');
                if (input) {
                    input.value = scannedCode;
                    filterBatchEquipmentList();
                }
            } else if (activeScanTarget === 'search') {
                switchTab('inventory');
                const searchInput = document.getElementById('filterSearch');
                if (searchInput) {
                    searchInput.value = scannedCode;
                    applyFilters();

                    const exactMatch = inventory.find(i => i.serial.toUpperCase() === scannedCode);
                    if (exactMatch && isWarehouse(exactMatch.status) && currentRole !== 'operator') {
                        showConfirm(
                            '¡Equipo Encontrado!',
                            `Serial: ${exactMatch.serial}\nTipo: ${exactMatch.type}\n\n¿Desea realizar el despacho / asignación inmediatamente?`,
                            () => openDispatchModal(exactMatch.id)
                        );
                    }
                }
            }
        }

        function scanFromFile(e) {
            const file = e.target.files[0];
            if (!file) return;

            const statusBox = document.getElementById('scannerStatus');
            statusBox.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> Procesando foto en alta definición...`;

            const html5QrCode = new Html5Qrcode("qr-reader");
            html5QrCode.scanFile(file, true)
                .then(decodedText => {
                    onScanSuccess(decodedText, null);

            try {
                const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(880, audioCtx.currentTime);

        function getStatusSortWeight(status) {
            if (isWarehouse(status)) return 1;
            if (isAssigned(status)) return 2;
            if (isMaintenance(status)) return 3;
            return 4;

        // 1. RESPALDO DE INVENTARIO (DEPÓSITO, ASIGNADO Y MANTENIMIENTO EN ORDEN DE EQUIPOS)
        function exportExcelInventory(format = 'xlsx') {
            if (!inventory || inventory.length === 0) {
                showToast('No hay equipos registrados para respaldar.', 'warning');
                return;
            }

            try {
                // Ordenar inventario: 1. Estado (Depósito, Asignado, Mantenimiento), 2. Tipo, 3. Marca/Modelo, 4. Serial
                const sorted = [...inventory].sort((a, b) => {
                    const wA = getStatusSortWeight(a.status);
                    const wB = getStatusSortWeight(b.status);
                    if (wA !== wB) return wA - wB;

                    const typeA = (a.type || '').toLowerCase();
                    const typeB = (b.type || '').toLowerCase();
                    if (typeA !== typeB) return typeA.localeCompare(typeB);

                    const brandA = (a.brand || '').toLowerCase();
                    const brandB = (b.brand || '').toLowerCase();
                    if (brandA !== brandB) return brandA.localeCompare(brandB);

                    return (a.serial || '').localeCompare(b.serial || '');
                });

                const mapItemToRow = (item, idx) => ({
                    "N°": idx + 1,
                    "Serial": item.serial || '',
                    "Tipo de Equipo": item.type || '',
                    "Marca": item.brand || '',
                    "Modelo": item.model || '',
                    "Procesador": item.processor || '-',
                    "Memoria RAM": item.ram || '-',
                    "Disco Duro": item.disk || '-',
                    "Color": item.color || '-',
                    "Estado": normalizeItemStatus(item.status),
                    "Agencia Asignada": item.agency && item.agency !== '-' ? item.agency : (isWarehouse(item.status) ? 'Depósito Principal' : '-'),
                    "Receptor / Responsable": item.receptor && item.receptor !== '-' ? item.receptor : '-',
                    "Dueño / Titular": item.owner && item.owner !== '-' ? item.owner : '-',
                    "Fecha Entrada (VE)": item.dateIn || '-',
                    "Fecha Salida (VE)": item.dateOut || '-',
                    "Observaciones": item.obs || '-'
                });

                const allRows = sorted.map(mapItemToRow);
                const warehouseRows = sorted.filter(i => isWarehouse(i.status)).map(mapItemToRow);
                const assignedRows = sorted.filter(i => isAssigned(i.status)).map(mapItemToRow);
                const maintRows = sorted.filter(i => isMaintenance(i.status)).map(mapItemToRow);

                const wb = XLSX.utils.book_new();

                // Hoja 1: Todo el inventario ordenado
                const wsAll = XLSX.utils.json_to_sheet(allRows);
                XLSX.utils.book_append_sheet(wb, wsAll, "Inventario General");

                // Hoja 2: En Depósito
                if (warehouseRows.length > 0) {
                    const wsDep = XLSX.utils.json_to_sheet(warehouseRows);
                    XLSX.utils.book_append_sheet(wb, wsDep, "En Depósito");
                }

                // Hoja 3: Asignados
                if (assignedRows.length > 0) {
                    const wsAsig = XLSX.utils.json_to_sheet(assignedRows);
                    XLSX.utils.book_append_sheet(wb, wsAsig, "Asignados en Agencias");
                }

                // Hoja 4: En Mantenimiento
                if (maintRows.length > 0) {
                    const wsMaint = XLSX.utils.json_to_sheet(maintRows);
                    XLSX.utils.book_append_sheet(wb, wsMaint, "En Mantenimiento");
                }

                const fileName = `Respaldo_Inventario_Equipos_${getFormattedDateOnly()}.${format}`;
                XLSX.writeFile(wb, fileName, { bookType: format === 'xls' ? 'biff8' : 'xlsx' });

                showToast(`Respaldo de inventario (${format.toUpperCase()}) descargado con éxito.`, 'success');
            } catch (err) {
                console.error("Error al exportar inventario a Excel:", err);
                showToast('Error al generar el archivo Excel de inventario.', 'error');
            }
        }

        // Sube el historial (1 documento por movimiento) por lotes, con progreso. Antes la importacion NO subia movimientos a la nube.
        async function uploadMovementsToCloud(list) {
            if (!window.db || !Array.isArray(list) || !list.length) return;
            const total = list.length; let done = 0;
            try {
                for (let i = 0; i < total; i += 25) {
                    const chunk = list.slice(i, i + 25).filter(m => m && m.id);
                    await Promise.all(chunk.map(m => window.setDoc(window.doc(window.db, "lotto_movements", String(m.id)), JSON.parse(JSON.stringify(m)))));
                    done += chunk.length;
                    showToast(`Subiendo historial a la nube: ${Math.min(done, total)} / ${total}`, 'info');
                }
                showToast(`Historial completo en la nube (${total} movimientos).`, 'success');
            } catch (err) {
                console.error('[Import] Error subiendo movimientos:', err);
                handleFirestoreError(err, 'uploadMovements');
            }
        }

        // Limpia un respaldo importado: sin seriales duplicados (ignorando espacios/simbolos), ids/codigos unicos y sin vacios criticos
        function sanitizeImportedData() {
            const score = (x) => (x.status === 'Asignado' ? 1000 : 0) + ((x.processor || '').trim() ? 100 : 0) + (String(x.id).startsWith('REC-') ? 0 : 50);
            const byKey = new Map();
            inventory.forEach(x => {
                if (!x) return;
                const k = x.serial && String(x.serial).trim() ? normSerialKey(x.serial) : ('ID_' + x.id);
                const cur = byKey.get(k);
                if (!cur || score(x) > score(cur) || (score(x) === score(cur) && String(x.dateIn || '') > String(cur.dateIn || ''))) byKey.set(k, x);
            });
            const removedInv = inventory.length - byKey.size;
            inventory = Array.from(byKey.values());
            const ids = new Set();
            inventory.forEach(x => {
                if (ids.has(String(x.id))) x.id = String(x.id) + '_' + normSerialKey(x.serial).slice(0, 6);
                ids.add(String(x.id));
                ['agency', 'receptor', 'owner'].forEach(f => { if (x[f] === '' || x[f] == null) x[f] = '-'; });
            });
            const usedId = new Set(), usedCode = new Set(); let fixedAg = 0;
            agencies = agencies.filter(a => a && a.name);
            agencies.forEach(a => {
                [['id', usedId], ['code', usedCode]].forEach(([f, used]) => {
                    let v = String(a[f] || '').trim();
                    if (!v) v = 'AG-' + Math.random().toString(36).slice(2, 6).toUpperCase();
                    let n = 2, nv = v;
                    while (used.has(nv.toUpperCase())) { nv = v + '-' + n++; fixedAg++; }
                    a[f] = nv; used.add(nv.toUpperCase());
                });
            });
            const seenMov = new Set();
            movements = movements.filter(m => m && m.id && !seenMov.has(m.id) && seenMov.add(m.id));
            return { removedInv, fixedAg };
        }

        function importJSONBackup(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    const data = JSON.parse(evt.target.result);
                    if (data.inventory && data.agencies) {
                        inventory = data.inventory;
                        agencies = data.agencies;
                        movements = data.movements || [];
                        maintenanceLog = data.maintenanceLog || [];
                        if (data.categories) equipmentCategories = data.categories;
                        if (data.stockThresholds) stockThresholds = data.stockThresholds;

                        if (typeof updateDeletedEquipmentFromMovements === 'function') updateDeletedEquipmentFromMovements();
                        if (typeof updateDeletedAgenciesFromMovements === 'function') updateDeletedAgenciesFromMovements();
                        if (typeof isEquipmentDeleted === 'function') {
                            inventory = inventory.filter(i => !isEquipmentDeleted(i.serial, i.id));
                        }
                        if (typeof isAgencyDeleted === 'function') {
                            agencies = agencies.filter(a => !isAgencyDeleted(a.id, a.code, a.name));
                        }

                        const _san = sanitizeImportedData();
                        saveData();
                        populateTypeDropdowns();
                        populateAgencyDropdowns();
                        renderAll();
                        closeModal('backupModal');
                        showToast(`¡Base de datos restaurada! ${inventory.length} equipos, ${agencies.length} agencias, ${movements.length} movimientos.` + (_san.removedInv ? ` (${_san.removedInv} duplicados depurados)` : ''));
                        uploadMovementsToCloud(movements);
                    } else {
                        showToast('El archivo de respaldo no tiene el formato correcto.', 'error');
                    }
                } catch (err) {
                    showToast('Error al leer el archivo JSON.', 'error');
                }
            };
            reader.readAsText(file);
        }

        // FUSIÓN MULTI-EQUIPO INTELIGENTE DE RESPALDOS JSON
        function mergeMultipleJSONBackups(e) {
            const files = Array.from(e.target.files);
            if (!files || files.length === 0) return;

            if (files.length === 1) {
                importJSONBackup(e);
                return;
            }

            const promises = files.map(file => {
                return new Promise((resolve, reject) => {
                    const reader = new FileReader();
                    reader.onload = (evt) => {
                        try {
                            const parsed = JSON.parse(evt.target.result);
                            resolve({ name: file.name, data: parsed });
                        } catch (err) {
                            reject(`Error al leer archivo ${file.name}`);
                        }
                    };
                    reader.onerror = () => reject(`Error de lectura en ${file.name}`);
                    reader.readAsText(file);
                });
            });

            Promise.all(promises).then(results => {
                const movMap = new Map();
                movements.forEach(m => { if (m && m.id) movMap.set(m.id, m); });

                const agMap = new Map();
                agencies.forEach(a => { if (a && a.name) agMap.set(agKey(a), a); });

                const maintMap = new Map();
                maintenanceLog.forEach(m => { if (m && m.id) maintMap.set(m.id, m); });

                const invMap = new Map();
                inventory.forEach(i => {
                    if (!i) return;
                    const key = (i.serial && i.serial.trim()) ? i.serial.trim().toUpperCase() : ('ID_' + i.id);
                    invMap.set(key, i);
                });

                let totalNewMovements = 0;
                let totalNewAgencies = 0;

                results.forEach(res => {
                    const d = res.data;
                    if (!d) return;

                    // Movimientos
                    if (Array.isArray(d.movements)) {
                        d.movements.forEach(m => {
                            if (m && m.id && !movMap.has(m.id)) {
                                movMap.set(m.id, m);
                                totalNewMovements++;
                            }
                        });
                    }

                    // Agencias
                    if (Array.isArray(d.agencies)) {
                        d.agencies.forEach(a => {
                            if (a && a.name) {
                                if (typeof isAgencyDeleted === 'function' && isAgencyDeleted(a.id, a.code, a.name)) return;
                                const k = agKey(a);
                                if (!agMap.has(k)) {
                                    agMap.set(k, a);
                                    totalNewAgencies++;
                                } else {
                                    const cur = agMap.get(k);
                                    if (!cur.phone && a.phone) cur.phone = a.phone;
                                    if (!cur.address && a.address) cur.address = a.address;
                                    if (!cur.owner && a.owner) cur.owner = a.owner;
                                }
                            }
                        });
                    }

                    // Mantenimiento
                    if (Array.isArray(d.maintenanceLog)) {
                        d.maintenanceLog.forEach(m => {
                            if (m && m.id && !maintMap.has(m.id)) maintMap.set(m.id, m);
                        });
                    }

                    // Equipos nuevos
                    if (Array.isArray(d.inventory)) {
                        d.inventory.forEach(item => {
                            if (!item) return;
                            const key = (item.serial && item.serial.trim()) ? item.serial.trim().toUpperCase() : ('ID_' + item.id);
                            if (!invMap.has(key)) {
                                invMap.set(key, item);
                            }
                        });
                    }
                });

                // Ordenar movimientos cronológicamente
                movements = Array.from(movMap.values()).sort((a, b) => (a.date || '').localeCompare(b.date || ''));
                agencies = Array.from(agMap.values());
                maintenanceLog = Array.from(maintMap.values());
                inventory = Array.from(invMap.values());

                // Actualizar el estado de cada equipo según su ÚLTIMO movimiento en el historial fusionado
                const lastMovBySerial = new Map();
                movements.forEach(mov => {
                    if (!mov || !Array.isArray(mov.items)) return;
                    mov.items.forEach(it => {
                        if (it && it.serial && it.serial !== '-') {
                            const s = it.serial.trim().toUpperCase();
                            lastMovBySerial.set(s, mov);
                        }
                    });
                });

                inventory.forEach(eq => {
                    if (!eq || !eq.serial) return;
                    const s = eq.serial.trim().toUpperCase();
                    if (lastMovBySerial.has(s)) {
                        const lastM = lastMovBySerial.get(s);
                        const t = (lastM.type || '').toLowerCase();
                        if (t.includes('despacho') || t.includes('asignación') || t.includes('combo')) {
                            eq.status = 'En Agencia';
                            if (lastM.agency) eq.agency = lastM.agency;
                            if (lastM.receptor) eq.receptor = lastM.receptor;
                            if (lastM.owner) eq.owner = lastM.owner;
                        } else if (t.includes('reingreso') || t.includes('retorno') || t.includes('depósito')) {
                            eq.status = 'En Depósito';
                            eq.agency = '';
                            eq.receptor = '';
                        } else if (t.includes('mantenimiento') || t.includes('taller')) {
                            eq.status = 'En Mantenimiento';
                        }
                    }
                });

                // Limpiar eliminados si los hay
                if (typeof updateDeletedEquipmentFromMovements === 'function') updateDeletedEquipmentFromMovements();
                if (typeof updateDeletedAgenciesFromMovements === 'function') updateDeletedAgenciesFromMovements();
                if (typeof isEquipmentDeleted === 'function') {
                    inventory = inventory.filter(i => !isEquipmentDeleted(i.serial, i.id));
                }
                if (typeof autoRecoverMissingEquipmentsFromMovements === 'function') autoRecoverMissingEquipmentsFromMovements();
                if (typeof isEquipmentDeleted === 'function') {
                    inventory = inventory.filter(i => !isEquipmentDeleted(i.serial, i.id));
                }
                if (typeof isAgencyDeleted === 'function') {
                    agencies = agencies.filter(a => !isAgencyDeleted(a.id, a.code, a.name));
                }

                saveData();
                populateTypeDropdowns();
                populateAgencyDropdowns();
                renderAll();
                closeModal('backupModal');

                // Descargar automáticamente el respaldo consolidado maestro
                const consolidatedData = {
                    inventory,
                    agencies,
                    movements,
                    maintenanceLog,
                    categories: equipmentCategories,
                    stockThresholds,
                    exportDate: getFormattedNow(),
                    mergedFrom: files.map(f => f.name)
                };
                const blob = new Blob([JSON.stringify(consolidatedData, null, 2)], { type: 'application/json' });
                const link = document.createElement("a");
                link.href = URL.createObjectURL(blob);
                link.download = `Respaldo_FUSION_CONSOLIDADA_${getFormattedDateOnly()}.json`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);

                showToast(`¡Fusión Multi-Equipo Exitosa! Se unieron ${files.length} respaldos: +${totalNewMovements} movimientos y +${totalNewAgencies} agencias integrados sin pérdida de datos.`, 'success');
            }).catch(err => {
                console.error(err);
                showToast(typeof err === 'string' ? err : 'Error al procesar la fusión de respaldos.', 'error');
            });
        }

        // CSV RÁPIDO DE INVENTARIO
        function exportData() {
            let csv = 'ID,Tipo,Marca,Modelo,Color,Procesador,RAM,DiscoDuro,Serial,Estado,Agencia,Receptor,Dueño,FechaEntrada,FechaSalida\n';
            inventory.forEach(i => {

            });

            const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement("a");
            const url = URL.createObjectURL(blob);
            link.setAttribute("href", url);
            link.setAttribute("download", `Inventario_GrupoAndino_${getFormattedDateOnly()}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            showToast('Archivo CSV exportado con compatibilidad UTF-8 (Excel).');
        }
